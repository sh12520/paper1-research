def utc():
    return datetime.now(timezone.utc)


def digest(path):
    with path.open('rb') as stream:
        return hashlib.file_digest(stream, 'sha256').hexdigest()


def atomic(path, contents):
    tmp = path.with_suffix(path.suffix + '.part')
    tmp.write_text(json.dumps(contents, ensure_ascii=False, indent=2, allow_nan=False) + '\n')
    os.replace(tmp, path)


def read(path):
    return json.loads(path.read_text()) if path.is_file() else {}


def script(name, *args, capture=False):
    return subprocess.run([sys.executable, '-B', '-u', str(ROOT / 'scripts' / name), *map(str, args)],
                          cwd=ROOT, text=True, check=True, capture_output=capture).stdout


def indexed_records(path):
    return sum(1 for _ in path.glob('[0-9][0-9][0-9][0-9].json'))


def frames(path):
    return indexed_records(path / 'per-frame')


def unit_state(unit):
    process = subprocess.run(['systemctl', '--user', 'show', f'{unit}.service', '-p', 'ActiveState',
                              '--value'], capture_output=True, text=True, check=False)
    return process.stdout.strip() or 'inactive'


def enqueue(unit, name, *args, nice=10):
    state = unit_state(unit)
    if state in {'active', 'activating', 'deactivating', 'reloading', 'failed'}:
        return state
    log = STATE / (unit + '.log')
    subprocess.run(['systemd-run', '--user', '--quiet', '--unit=' + unit, f'--property=Nice={nice}',
                    '--property=WorkingDirectory=' + str(ROOT), '--property=StandardOutput=append:' + str(log),
                    '--property=StandardError=inherit', '--setenv=OMP_NUM_THREADS=2',
                    '--setenv=PYTHONDONTWRITEBYTECODE=1', sys.executable, '-B', '-u', str(ROOT / 'scripts' / name),
                    *map(str, args)], check=True)
    return 'queued'


def current_coverage():
    completed, extra = frames(FORMAL), frames(CONTROL)
    prior = read(STATE / 'status.json').get('coverage', {})
    prior_summary = ROOT / prior.get('summary', '')
    latest = sorted(RESULTS.glob('coverage-*/coverage.json'), reverse=True)
    latest_relevant = read(latest[0]) if latest else {}
    if prior.get('source_frame_counts') == [completed, extra] and 'status' in prior and prior_summary.is_file() and (
            not latest or prior_summary == latest[0]):
        return prior
    formal_counts = latest_relevant.get('frame_counts', {})
    control_counts = latest_relevant.get('exploratory', {}).get('frame_counts', {})
    if latest and sum(formal_counts.get(key, 0) for key in ('audited', 'invalid')) == completed and (
            sum(control_counts.get(key, 0) for key in ('audited', 'invalid')) == extra):
        summary = latest[0]
    else:
        result = subprocess.run([sys.executable, '-B', str(ROOT / 'scripts/opv2v_autoresearch_coverage.py')],
                                cwd=ROOT, text=True, capture_output=True)
        if result.returncode not in (0, 2) or not result.stdout.strip():
            raise RuntimeError(result.stderr.strip() or 'Coverage tool failed before saving a receipt')
        summary = Path(json.loads(result.stdout.splitlines()[-1])['snapshot']) / 'coverage.json'
    contents = read(summary)
    return {'source_frame_counts': [completed, extra], 'summary': str(summary.relative_to(ROOT)),
            'summary_sha256': digest(summary), 'frame_counts': contents['frame_counts'],
            'frozen_schedule_detections': contents['frozen_schedule_detections'],
            'status': contents['status'],
            'exploratory': contents.get('exploratory'),
            'formal_coverage_complete': contents['formal_coverage_complete']}

def advance_r23(details):
    cpu = R23 / 'cpu-smoke/completion.json'
    smoke = R23 / 'smoke/completion.json'
    screen = R23 / 'screen/completion.json'
    report = R23 / 'report.json'
    budget = R23 / 'deadline-budget.json'
    nms = R23 / 'nms-probe/completion.json'
    incremental = R23 / 'incremental-control.json'
    core = ROOT / 'scripts/opv2v_r23_prefilter.py'
    runner = ROOT / 'scripts/run_opv2v_r23.py'
    reporter = ROOT / 'scripts/report_opv2v_r23.py'
    details.update(r23_protocol_sha256=digest(R23_PROTOCOL), r23_frames=frames(screen.parent),
                   r23_full_input_frames=indexed_records(screen.parent / 'full-validation'))
    for label, path in (('cpu_acceptance', cpu), ('gpu_acceptance', smoke), ('panel_completion', screen), ('report', report)):
        if path.is_file():
            details['r23_' + label] = str(path.relative_to(ROOT))
    evidence = read(report)
    if evidence.get('status') == 'failed_equivalence':
        details['r23_decision'] = evidence.get('decision')
        return 'r23_equivalence_failed', details
    if evidence.get('status') == 'complete':
        details['r23_decision'] = evidence.get('decision')
        if read(budget).get('paired_report_sha256') != digest(report):
            details['r23_deadline_budget_worker'] = enqueue('research-opv2v-r23-deadline-budget',
                                                           'report_opv2v_r23_deadline_budget.py', nice=15)
            return 'r23_deadline_budget', details
        details['r23_deadline_budget'] = str(budget.relative_to(ROOT))
        stages = evidence['timing']['per_scope']['cooperative']
        post = stages.get('optimized.measurement.untraced_cpu_postprocess_wall_ms', {}).get('p50', 0)
        nms_wall = stages.get('optimized.traced_cpu_stage_total_ms.rotated_nms', {}).get('p50', 0)
        ratio = nms_wall / post if post else 0
        details['r23_residual_nms_to_postprocess_p50_ratio'] = ratio
        if ratio >= .30 and details['r23_decision'].get('largest_profiled_residual_cpu_postprocess_component') == 'rotated_nms':
            if read(nms).get('paired_r23_report_sha256') == digest(report):
                details['r23_residual_nms_probe'] = str(nms.relative_to(ROOT))
            else:
                nms_runner = ROOT / 'scripts/run_opv2v_r23_nms_probe.py'
                if nms_runner.is_file():
                    details['r23_residual_nms_worker'] = enqueue('research-opv2v-r23-nms-probe', nms_runner.name, nice=15)
                return 'r23_residual_cpu_nms', details
        if details['r23_decision'].get('next_focus') == 'validate_incremental_fusion_compute_budget':
            if read(incremental).get('primary_report_sha256') == digest(report):
                details['r23_incremental_control'] = str(incremental.relative_to(ROOT))
            else:
                control = ROOT / 'scripts/report_opv2v_r23_incremental_control.py'
                if control.is_file():
                    details['r23_incremental_control_worker'] = enqueue('research-opv2v-r23-incremental-control',
                                                                      control.name, nice=15)
                return 'r23_incremental_control', details
        if R24_PROTOCOL.is_file():
            return advance_r24(details)
        return 'r23_complete', details
    if screen.is_file():
        if reporter.is_file():
            details['r23_report_worker'] = enqueue('research-opv2v-r23-report', reporter.name, nice=15)
        return 'r23_analysis', details
    if not cpu.is_file() or not core.is_file() or not runner.is_file():
        if (R23 / 'cpu-smoke/failure.json').is_file():
            details['r23_cpu_mismatch'] = str((R23 / 'cpu-smoke/failure.json').relative_to(ROOT))
            return 'r23_cpu_acceptance_blocked', details
        return 'r23_prepare_cpu_acceptance', details
    proof = read(cpu)
    if (proof.get('status') != 'complete' or proof.get('all_exact') is not True or proof.get('states_verified') != 4
            or proof.get('core_sha256') != digest(core)):
        return 'r23_cpu_acceptance_blocked', details
    if smoke.is_file():
        accepted = read(smoke)
        smoke_run = smoke.parent / 'run.json'
        hashes = read(smoke_run).get('scripts_sha256', {})
        if (accepted.get('status') != 'verified' or accepted.get('all_frames_verified') is not True
                or not smoke_run.is_file() or accepted.get('run_sha256') != digest(smoke_run)
                or hashes.get(core.name) != digest(core) or hashes.get(runner.name) != digest(runner)):
            return 'r23_gpu_acceptance_failed', details
        details['r23_panel_worker'] = enqueue('research-opv2v-r23-panel', 'autoresearch_opv2v.py', 'r23-panel')
        return 'r23_panel', details
    details['r23_gpu_acceptance_worker'] = enqueue('research-opv2v-r23-smoke', 'autoresearch_opv2v.py', 'r23-smoke')
    return 'r23_gpu_acceptance', details


def advance_r24(details):
    screen = R24 / 'screen/completion.json'
    dispatch = R24 / 'dispatch-ready.json'
    report = R24 / 'report.json'
    followup = R24 / 'mechanism.json'
    rules = ROOT / 'scripts/opv2v_r24_schedule.py'
    runner = ROOT / 'scripts/run_opv2v_r24.py'
    reporter = ROOT / 'scripts/report_opv2v_r24.py'
    details.update(r24_protocol_sha256=digest(R24_PROTOCOL), r24_frames=frames(screen.parent))
    for label, path in (('dispatch', dispatch), ('panel_completion', screen), ('report', report), ('mechanism', followup)):
        if path.is_file():
            details['r24_' + label] = str(path.relative_to(ROOT))
    evidence = read(report)
    if evidence.get('status') == 'complete':
        details['r24_decision'] = evidence.get('decision')
        if read(followup).get('primary_report_sha256') == digest(report):
            if R25_PROTOCOL.is_file():
                return advance_r25(details)
            return 'r24_complete', details
        script_path = ROOT / 'scripts/report_opv2v_r24_mechanism.py'
        if script_path.is_file():
            details['r24_mechanism_worker'] = enqueue('research-opv2v-r24-mechanism', script_path.name, nice=15)
        return 'r24_mechanism', details
    if evidence.get('status') in {'failed_exactness', 'failed_equivalence', 'mismatch'}:
        details['r24_decision'] = evidence.get('decision')

def advance(coverage):
    formal = unit_state('research-opv2v-evaluate')
    if coverage.get('status') == 'invalid_formal_artifacts' or coverage['frame_counts']['invalid']:
        return 'formal_artifact_review', {'formal_worker': formal}
    if not coverage['formal_coverage_complete']:
        return 'formal_inference', {'formal_worker': enqueue('research-opv2v-evaluate', 'complete_opv2v_evaluation.py')}
    if not formal_complete():
        if formal in {'active', 'activating', 'deactivating'}:
            return 'formal_statistics', {'formal_worker': formal}
        return 'formal_statistics', {'report_worker': enqueue('research-opv2v-formal-reports', 'autoresearch_opv2v.py', 'formal-reports')}
    if not (CONTROL_SMOKE / 'verification.json').is_file():
        return 'control_verification', {'verification_worker': enqueue('research-opv2v-control-verification',
                'run_opv2v_feedback_control.py', CONTROL_SMOKE, '--reference', FORMAL, '--verify-smoke', '--gpu', 'cpu')}
    smoke = read(CONTROL_SMOKE / 'verification.json')
    if (smoke.get('smoke_excluded') is not True
            or smoke.get('runner_sha256') != digest(ROOT / 'scripts/run_opv2v_feedback_control.py')
            or smoke.get('reference_sha256') != digest(ROOT / 'results/opv2v-budget-guard-smoke/verification.json')
            or not smoke.get('record', {}).get('exact_application_packet_roundtrips')):
        return 'control_provenance_review', {'verification_receipt': str((CONTROL_SMOKE / 'verification.json').relative_to(ROOT))}
    if not (CONTROL / 'detections.jsonl').is_file():
        return 'matched_feedback_control', {'control_worker': enqueue('research-opv2v-feedback-control',
                'autoresearch_opv2v.py', 'control-run')}
    if (coverage['exploratory'] or {}).get('coverage_complete') is not True:
        return 'control_artifact_review', {'artifact_status': coverage['exploratory']}
    report = RESULTS / 'feedback-attribution.json'
    details = {'bilateral_prototype': 'HOLD'}
    if not report.is_file():
        details['attribution_worker'] = enqueue('research-opv2v-feedback-attribution',
                                               'report_opv2v_feedback_attribution.py', '--output', report)
    abc = RESULTS / 'marginal-abc.json'
    if not abc.is_file():
        details['marginal_abc_worker'] = enqueue('research-opv2v-marginal-abc', 'report_opv2v_marginal.py',
                                                '--abc-only', '--output', abc)
    else:
        details['marginal_abc'] = str(abc.relative_to(ROOT))
    bc_targets = RESULTS / 'r1d-bc-target-transitions/summary.json'
    if bc_targets.is_file():
        details['r1d_bc_target_transitions'] = str(bc_targets.relative_to(ROOT))
    elif abc.is_file() and report.is_file() and (ROOT / 'scripts/report_opv2v_bc_transitions.py').is_file():
        details['r1d_bc_target_worker'] = enqueue('research-opv2v-r1d-bc-transitions',
            'report_opv2v_bc_transitions.py')
    clock_only = RESULTS / 'feedback-clock-only.json'
    if not clock_only.is_file():
        details['feedback_clock_worker'] = enqueue('research-opv2v-feedback-clock',
                                                   'report_opv2v_marginal_followup.py', '--clock-only', '--output', clock_only)
    else:
        details['feedback_clock_evidence'] = str(clock_only.relative_to(ROOT))
    sender_deadline = RESULTS / 'ordinary-sender-deadline.json'
    if sender_deadline.is_file():
        details['ordinary_sender_deadline'] = str(sender_deadline.relative_to(ROOT))
    elif report.is_file():
        feedback_contrast = read(report)['settings'][PRIMARY]['comparisons']['parallel_minus_parallel_no_feedback']
        if feedback_contrast['interval_95']['upper']['ap70'] <= 0:
            details['ordinary_sender_deadline_worker'] = enqueue('research-opv2v-ordinary-deadline',
                'report_opv2v_marginal_followup.py', '--sender-deadline-only', '--output', sender_deadline)
    d_smoke = read(OPTIMIZED / 'smoke/verification.json')
    executable = ROOT / 'scripts/run_opv2v_sender_optimized.py'
    if (d_smoke.get('smoke_excluded') is not True or not executable.is_file()
            or d_smoke.get('executable_identity', {}).get(executable.name) != digest(executable)
            or d_smoke.get('control_sha256') != digest(CONTROL_SMOKE / 'verification.json')):
        return 'sender_optimized_verification', details
    b35_plan = ORDINARY_35 / 'plan.json'
    b35_smoke = ORDINARY_35 / 'smoke/verification.json'
    if b35_plan.is_file():
        plan = read(b35_plan)
        details['ordinary_35_plan'] = str(b35_plan.relative_to(ROOT))
        details['ordinary_35_frames'] = frames(ORDINARY_35)
        details['ordinary_35_cached_before_gpu'] = plan['frame_count'] - plan['counts']['needs_new_gpu_fusion']
        details['d_selection_frames'] = indexed_records(ORDINARY_35 / 'd-selection')
    details['sender_optimized_frames'] = frames(OPTIMIZED)
    if not (OPTIMIZED / 'detections.jsonl').is_file():
        details['sender_optimized_worker'] = enqueue('research-opv2v-sender-optimized', 'autoresearch_opv2v.py', 'optimized-run')
        return 'sender_optimized', details
    if b35_plan.is_file():
        if not (ORDINARY_35 / 'completion.json').is_file() and b35_smoke.is_file():
            details['ordinary_35_worker'] = enqueue('research-opv2v-r1d-ordinary35',
                'autoresearch_opv2v.py', 'ordinary-35-run')
        elif (ORDINARY_35 / 'completion.json').is_file():
            b35_report = RESULTS / 'r1d-ordinary-35-complete.json'
            if not b35_report.is_file():
                details['ordinary_35_report_worker'] = enqueue('research-opv2v-r1d-ordinary35-report',
                    'report_opv2v_marginal_followup.py', '--sender-deadline-complete', '--output', b35_report)
            else:
                details['ordinary_35_complete'] = str(b35_report.relative_to(ROOT))
            selected = ORDINARY_35 / 'd-selection/completion.json'
            if not selected.is_file() and unit_state('research-opv2v-r1d-ordinary35') not in {'active', 'activating'}:
                details['d_selection_worker'] = enqueue('research-opv2v-r1d-d-selection',
                    'autoresearch_opv2v.py', 'd-selection-run')
            elif selected.is_file():
                details['d_selection_receipt'] = str(selected.relative_to(ROOT))
    abcd = RESULTS / 'marginal-abcd.json'
    if not abcd.is_file():
        details['marginal_worker'] = enqueue('research-opv2v-marginal-abcd', 'report_opv2v_marginal.py', '--output', abcd)
        return 'four_group_marginal_statistics', details
    details['marginal_report'] = str(abcd.relative_to(ROOT))
    next_r1d = RESULTS / 'r1d-next.json'
    next_script = 'report_opv2v_r1d_next.py'
    if next_r1d.is_file():
        details['r1d_next_experiment'] = str(next_r1d.relative_to(ROOT))
    elif (ROOT / 'scripts' / next_script).is_file():
        details['r1d_next_worker'] = enqueue('research-opv2v-r1d-next', next_script, '--output', next_r1d)
    downstream = RESULTS / 'marginal-deadline-followup.json'
    if not downstream.is_file():
        details['deadline_followup_worker'] = enqueue('research-opv2v-marginal-deadline',
                                                       'report_opv2v_marginal_followup.py', '--output', downstream)
        return 'deadline_mechanism_followup', details
    phase = 'r1d_paired_mechanisms' if next_r1d.is_file() else 'r1d_branch_experiment'
    if 'ordinary_35_worker' in details or 'ordinary_35_report_worker' in details:
        phase = 'r1d_ordinary35_completion'
    if 'd_selection_worker' in details and 'ordinary_35_complete' in details:
        phase = 'r1d_d_selection_export'
    if next_r1d.is_file() and 'ordinary_35_complete' in details and 'd_selection_receipt' in details:
        phase = 'r1d_complete'
    decision = read(next_r1d).get('decision') if next_r1d.is_file() else read(abcd).get('decision')
    details.update(next_scientific_test=decision, deadline_followup=str(downstream.relative_to(ROOT)))
    if phase == 'r1d_complete' and R2_PROTOCOL.is_file():
        return advance_r2(details)
    return phase, details


def supervise():
    for path in (STATE, RESULTS, HISTORY):
        path.mkdir(parents=True, exist_ok=True)
    with (STATE / '.supervisor.lock').open('a') as lock:
        try:
            fcntl.flock(lock, fcntl.LOCK_EX | fcntl.LOCK_NB)
        except BlockingIOError:
            return
        coverage = current_coverage()
        phase, details = advance(coverage)
        stable = {'phase': phase, 'coverage': coverage, **details, 'protocol_sha256': digest(PROTOCOL)}
        for label, artifact in (
                ('archived_first_stage', RESULTS / 'first-only-archive-reuse-0397.json'),
                ('control_verification', CONTROL_SMOKE / 'verification.json')):
            if artifact.is_file():
                stable[label] = {'path': str(artifact.relative_to(ROOT)), 'sha256': digest(artifact)}
        path = STATE / 'status.json'
        before = read(path)
        if {key: value for key, value in before.items() if key != 'updated_utc'} != stable:
            now = utc()
            record = {**stable, 'updated_utc': now.isoformat()}
            atomic(HISTORY / (now.strftime('%Y%m%dT%H%M%S%fZ') + '.json'), record)
            atomic(path, record)
        print(f"{phase}, {coverage['frame_counts']['audited']}/2169 audited formal frames", flush=True)


def formal_reports():
    # A reboot after the last frame can land here with no active accelerator.
    with (FORMAL / '.run.lock').open('a') as lock:
        fcntl.flock(lock, fcntl.LOCK_EX | fcntl.LOCK_NB)
        result = script('opv2v_autoresearch_coverage.py', capture=True)
        summary = Path(json.loads(result.splitlines()[-1])['snapshot']) / 'coverage.json'
        if not read(summary)['formal_coverage_complete']:
            raise RuntimeError('CPU reporting requires all audited formal frames')
        target = FORMAL / 'detections.jsonl'
        if not target.is_file():
            tmp = target.with_suffix('.jsonl.part')
            with tmp.open('w') as stream:
                for index, line in enumerate(PANEL.read_text().splitlines()):
                    record = read(FORMAL / 'per-frame' / f'{index:04d}.json')
                    if record.get('frame_key') != json.loads(line)['frame_key']:
                        raise RuntimeError('Frozen frame changed after audit')
                    for row in record['detections']:
                        stream.write(json.dumps(row, allow_nan=False) + '\n')
            os.replace(tmp, target)
        script('complete_opv2v_evaluation.py', '--reports-only')
